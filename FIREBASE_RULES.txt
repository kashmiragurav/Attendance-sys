# Firebase Security Architecture
# Project: attendance-3519f
#
# IMPORTANT: This app uses the Firestore REST API with an API key only.
# Firebase Auth tokens are NOT used, so Firestore security rules below
# are NOT enforced by Firebase on incoming requests.
#
# App-level security boundaries (enforced in code):
#
# 1. writeAttendanceRecord() in firebaseConfig.js:
#    - Rejects writes where record.userId !== authenticated user's uid
#    - Rejects writes where record.companyId !== authenticated user's companyId
#    - Rejects writes to doc IDs that belong to a different user
#
# 2. AdminEditAttendanceScreen.handleSave():
#    - Verifies record.companyId === admin's companyId before any write
#    - Uses admin's verified companyId on new records (not client-supplied)
#
# 3. All attendance reads use getTodayAttendanceDoc(userId, date):
#    - Reads by deterministic doc ID att_{userId}_{date}
#    - No cross-tenant data is fetched
#
# 4. getUserAttendance(userId) uses where('userId', '==', userId):
#    - Scoped to the authenticated user's own records
#
# 5. Admin queries use where('companyId', '==', user.companyId):
#    - Scoped to the admin's own company
#
# 6. Role routing in App.js:
#    - SUPER_ADMIN → SuperAdminStack (company management only)
#    - admin/COMPANY_ADMIN → AdminStack (company-scoped)
#    - employee → AppStack (own data only)
#
# TO UPGRADE TO FULL FIREBASE AUTH SECURITY:
# 1. Migrate to Firebase Auth (email/password or custom tokens)
# 2. Pass the Firebase ID token in Authorization header on all REST calls
# 3. Deploy the Firestore rules below — they will then be enforced server-side
#
# ─────────────────────────────────────────────────────────────────────────────
# FIRESTORE RULES (enforced only when Firebase Auth tokens are used)
# ─────────────────────────────────────────────────────────────────────────────

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isAuthenticated() {
      return request.auth != null;
    }

    function getUserData() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }

    function isSuperAdmin() {
      return isAuthenticated() && getUserData().role == 'SUPER_ADMIN';
    }

    function isCompanyAdmin() {
      return isAuthenticated() &&
        (getUserData().role == 'admin' || getUserData().role == 'COMPANY_ADMIN');
    }

    function isOwnUser(userId) {
      return request.auth.uid == userId;
    }

    function sameCompany(companyId) {
      return isAuthenticated() && getUserData().companyId == companyId;
    }

    // ── Users ──────────────────────────────────────────────────────────────
    match /users/{userId} {
      allow read: if isOwnUser(userId) || isSuperAdmin() ||
                     (isCompanyAdmin() && sameCompany(resource.data.companyId));
      allow write: if isOwnUser(userId) || isSuperAdmin();
    }

    // ── Attendance ─────────────────────────────────────────────────────────
    match /attendance/{attendanceId} {
      // Read: own record, company admin of same company, or super admin
      allow read: if isAuthenticated() && (
        isOwnUser(resource.data.userId) ||
        isSuperAdmin() ||
        (isCompanyAdmin() && sameCompany(resource.data.companyId))
      );

      // Employee self-write: userId must match auth uid, companyId must match
      allow create: if isAuthenticated() &&
        isOwnUser(request.resource.data.userId) &&
        request.resource.data.companyId == getUserData().companyId;

      // Employee update own record (punch out, breaks)
      allow update: if isAuthenticated() &&
        isOwnUser(resource.data.userId) &&
        resource.data.companyId == getUserData().companyId;

      // Admin correction: same company only
      allow write: if isCompanyAdmin() &&
        sameCompany(request.resource.data.companyId);

      // Super admin: unrestricted
      allow write: if isSuperAdmin();
    }

    // ── Office Settings ────────────────────────────────────────────────────
    match /office_settings/{docId} {
      allow read: if isAuthenticated();
      allow write: if isSuperAdmin() || isCompanyAdmin();
    }

    // ── Companies ──────────────────────────────────────────────────────────
    match /companies/{companyId} {
      allow read: if isAuthenticated() && (
        isSuperAdmin() || getUserData().companyId == companyId
      );
      allow write: if isSuperAdmin();
    }

    // ── Catch-all: deny by default ─────────────────────────────────────────
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
